# Copies KiCad demo projects verbatim from a KiCad source checkout at the pinned

rm -rf "$DEST"; mkdir -p "$DEST/ecc83" "$DEST/complex_hierarchy"

cp "$SRC"/demos/ecc83/ecc83-pp.* "$DEST/ecc83/"

cp "$SRC"/demos/ecc83/ecc83-pp.kicad_sym "$DEST/ecc83/" 2>/dev/null || true
cp -r "$SRC"/demos/complex_hierarchy/. "$DEST/complex_hierarchy/"
find "$DEST" \( -name '*-backups' -o -name '*.lck' -o -name '*.kicad_prl' \) -prune -exec rm -rf {} +

export LANG=C LC_ALL=C
for f in $(find "$DEST" -name '*.kicad_sch'); do kicad-cli sch upgrade --force "$f"; done
for f in $(find "$DEST" -name '*.kicad_pcb'); do kicad-cli pcb upgrade --force "$f"; done

from __future__ import annotations

from typing import Any

NM_PER_MM = 1_000_000


def nm_to_mm(nm: int | float) -> float:
    return round(nm / NM_PER_MM, 4)


def mm_to_nm(mm: float) -> int:
    return round(mm * NM_PER_MM)


def pos_mm(v: Any) -> dict[str, float]:
    return {"x": nm_to_mm(v.x), "y": nm_to_mm(v.y)}

from __future__ import annotations

import difflib
from collections.abc import Iterable
from fnmatch import fnmatchcase

from copperchat.errors import StructuredToolError


def matches(pattern: str | None, *fields: str) -> bool:
    if not pattern:
        return True
    p = pattern.lower()
    return any(fnmatchcase(f.lower(), p) for f in fields if f)


def not_found(kind: str, name: str, candidates: Iterable[str]) -> StructuredToolError:
    pool = sorted(set(candidates))
    close = difflib.get_close_matches(name, pool, n=5, cutoff=0.5)
    suggestion = (f"Did you mean: {', '.join(close)}?" if close
                  else f"List the available {kind.lower()}s first (filter with a glob like 'R*').")
    return StructuredToolError(f"{kind} {name} not found",
                               f"No {kind.lower()} with that name exists in the open document",
                               suggestion)
```

from __future__ import annotations

import colorsys
import hashlib

# theme -> (lightness, saturation); "agent" renders on a dark background.
_THEMES: dict[str, tuple[float, float]] = {"agent": (0.62, 0.75), "dark": (0.62, 0.75),
                                           "light": (0.38, 0.80)}


def net_color(net_name: str, theme: str) -> tuple[int, int, int]:
    """Deterministic net colour: sha1(name) -> hue; lightness depends on the theme."""
    digest = hashlib.sha1(net_name.encode("utf-8")).digest()
    hue = int.from_bytes(digest[:2], "big") / 65536.0
    lightness, saturation = _THEMES.get(theme, _THEMES["agent"])
    r, g, b = colorsys.hls_to_rgb(hue, lightness, saturation)
    return round(r * 255), round(g * 255), round(b * 255)
```

from __future__ import annotations

from typing import Any

from copperchat.kicad_tools.units import nm_to_mm, pos_mm


def kiid(item: Any) -> str:
    return str(item.id.value)


def field_text(field: Any) -> str:
    return str(field.text.value) if hasattr(field.text, "value") else str(field.text)


def symbol_dict(sym: Any) -> dict[str, Any]:
    fields = {str(f.name): field_text(f) for f in sym.definition.fields} \
        if hasattr(sym.definition, "fields") else {}
    return {
        "uuid": kiid(sym),
        "ref": field_text(sym.reference_field),
        "value": field_text(sym.value_field),
        "footprint": field_text(sym.footprint_field),
        "position_mm": pos_mm(sym.position),
        "sheet": str(sym.path) if hasattr(sym, "path") else "/",
        "fields": fields,
    }


def footprint_dict(fp: Any, board: Any) -> dict[str, Any]:
    return {
        "uuid": kiid(fp),
        "ref": field_text(fp.reference_field),
        "value": field_text(fp.value_field),
        "footprint": str(fp.definition.id) if hasattr(fp.definition, "id") else "",
        "position_mm": pos_mm(fp.position),
        "rotation_deg": round(float(fp.orientation.degrees), 3),
        "side": "back" if board.get_layer_name(fp.layer).startswith("B.") else "front",
    }


def pad_dict(pad: Any) -> dict[str, Any]:
    return {"uuid": kiid(pad), "number": str(pad.number), "net": str(pad.net.name),
            "position_mm": pos_mm(pad.position)}


def track_dict(t: Any, board: Any) -> dict[str, Any]:
    return {"uuid": kiid(t), "start_mm": pos_mm(t.start), "end_mm": pos_mm(t.end),
            "width_mm": nm_to_mm(t.width), "layer": board.get_layer_name(t.layer)}


def via_dict(v: Any) -> dict[str, Any]:
    return {"uuid": kiid(v), "position_mm": pos_mm(v.position)}

def _info(h: KicadHandle) -> dict[str, Any]:
    out: dict[str, Any] = {"kicadVersion": str(h.kicad.get_version().full_version)}
    sch = h.schematic()
    out["sheets"] = [{"name": str(s.name), "path": str(s.path)} for s in sch.get_hierarchy()]

        x0 = min(b.pos.x for b in boxes); y0 = min(b.pos.y for b in boxes)
        x1 = max(b.pos.x + b.size.x for b in boxes); y1 = max(b.pos.y + b.size.y for b in boxes)

from __future__ import annotations

from typing import Any

from copperchat.kicad_session import KicadHandle
from copperchat.kicad_tools.access import field_text, kiid, symbol_dict
from copperchat.kicad_tools.filtering import matches, not_found
from copperchat.kicad_tools.units import pos_mm
from copperchat.tools.spec import ToolContext, ToolResult


def _symbols(h: KicadHandle) -> list[dict[str, Any]]:
    out = [symbol_dict(s) for s in h.schematic().get_symbols()]
    return sorted(out, key=lambda s: (s["ref"], s["uuid"]))


def _netlist(h: KicadHandle) -> dict[str, list[dict[str, Any]]]:
    nets: dict[str, list[dict[str, Any]]] = {}
    for net in h.schematic().get_netlist():
        pins: list[dict[str, Any]] = []
        for sheet in net.sheets:
            for item in sheet.items:   # pins on this sheet; 1C confirms the field name
                pins.append({"ref": field_text(item.parent_reference)
                             if hasattr(item, "parent_reference") else "",
                             "pin": str(getattr(item, "number", "")),
                             "uuid": kiid(item),
                             "position_mm": pos_mm(item.position),
                             "sheet": str(sheet.path)})
        nets[str(net.name)] = sorted(pins, key=lambda p: (p["ref"], p["pin"]))
    return nets


async def sch_list_symbols(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    pattern = args.get("filter")
    syms = [s for s in await ctx.session.call(_symbols)
            if matches(pattern, s["ref"], s["value"], s["footprint"])]
    return ToolResult(data={"count": len(syms), "symbols": syms})


async def sch_get_symbol(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    ref = str(args["ref"])
    syms = await ctx.session.call(_symbols)
    hit = next((s for s in syms if s["ref"] == ref), None)
    if hit is None:
        raise not_found("Symbol", ref, (s["ref"] for s in syms))
    nets = await ctx.session.call(_netlist)
    hit["pins"] = sorted(({"pin": p["pin"], "net": name, "uuid": p["uuid"],
                           "position_mm": p["position_mm"]}
                          for name, pins in nets.items() for p in pins if p["ref"] == ref),
                         key=lambda p: p["pin"])
    return ToolResult(data=hit)


async def sch_get_net(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    name = str(args["name"])
    nets = await ctx.session.call(_netlist)
    if name not in nets:
        raise not_found("Net", name, nets.keys())
    return ToolResult(data={"name": name, "pins": nets[name]})

from __future__ import annotations

from collections import defaultdict
from typing import Any

from copperchat.kicad_session import KicadHandle
from copperchat.kicad_tools.access import (footprint_dict, kiid, pad_dict, track_dict,
                                           via_dict)
from copperchat.kicad_tools.filtering import matches, not_found
from copperchat.kicad_tools.units import pos_mm
from copperchat.tools.spec import ToolContext, ToolResult


def board_snapshot(h: KicadHandle) -> dict[str, Any]:
    """Everything the tools and the agent renderer need, in mm, in one worker call."""
    board = h.board()
    fps = []
    for fp in board.get_footprints():
        d = footprint_dict(fp, board)
        d["pads"] = sorted((pad_dict(p) for p in fp.definition.pads), key=lambda p: p["number"])
        bbox = board.get_item_bounding_box(fp)
        d["bbox_mm"] = {"x": bbox.pos.x / 1e6, "y": bbox.pos.y / 1e6,
                        "w": bbox.size.x / 1e6, "h": bbox.size.y / 1e6}
        fps.append(d)
    tracks = [track_dict(t, board) | {"net": str(t.net.name)} for t in board.get_tracks()
              if hasattr(t, "start")]
    vias = [via_dict(v) | {"net": str(v.net.name)} for v in board.get_vias()]
    outline = [{"start_mm": pos_mm(s.start), "end_mm": pos_mm(s.end)}
               for s in board.get_shapes()
               if board.get_layer_name(s.layer) == "Edge.Cuts" and hasattr(s, "end")]
    # Unrouted: pads of a net that are not in the same connected copper cluster.
    by_net: dict[str, list[Any]] = defaultdict(list)
    for p in board.get_pads():
        if p.net.name:
            by_net[str(p.net.name)].append(p)
    unconnected = []
    for net, pads in by_net.items():
        seen: set[str] = set()
        clusters: list[list[Any]] = []
        for p in pads:
            if kiid(p) in seen:
                continue
            connected = {kiid(i) for i in board.get_connected_items(p)} | {kiid(p)}
            cluster = [q for q in pads if kiid(q) in connected]
            seen |= {kiid(q) for q in cluster}
            clusters.append(cluster)
        for a, b in zip(clusters, clusters[1:], strict=False):
            unconnected.append({"net": net, "from_mm": pos_mm(a[0].position),
                                "to_mm": pos_mm(b[0].position)})
    return {"footprints": sorted(fps, key=lambda f: (f["ref"], f["uuid"])), "tracks": tracks,
            "vias": vias, "outline": outline, "unconnected": unconnected,
            "nets": sorted(str(n.name) for n in board.get_nets() if n.name)}


async def pcb_list_footprints(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    snap = await ctx.session.call(board_snapshot)
    fps = [{k: v for k, v in f.items() if k not in ("pads", "bbox_mm")}
           for f in snap["footprints"]
           if matches(args.get("filter"), f["ref"], f["value"], f["footprint"])]
    return ToolResult(data={"count": len(fps), "footprints": fps})


async def pcb_get_footprint(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    ref = str(args["ref"])
    snap = await ctx.session.call(board_snapshot)
    hit = next((f for f in snap["footprints"] if f["ref"] == ref), None)
    if hit is None:
        raise not_found("Footprint", ref, (f["ref"] for f in snap["footprints"]))
    return ToolResult(data={k: v for k, v in hit.items() if k != "bbox_mm"})


async def pcb_get_net(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    name = str(args["name"])
    snap = await ctx.session.call(board_snapshot)
    if name not in snap["nets"]:
        raise not_found("Net", name, snap["nets"])
    pads = [{"ref": f["ref"], **p} for f in snap["footprints"] for p in f["pads"]
            if p["net"] == name]
    return ToolResult(data={
        "name": name, "pads": pads,
        "tracks": [t for t in snap["tracks"] if t["net"] == name],
        "vias": [v for v in snap["vias"] if v["net"] == name],
        "unconnected": [u for u in snap["unconnected"] if u["net"] == name]})

from __future__ import annotations

import asyncio
import json
import os
import tempfile
from pathlib import Path

from copperchat.config import Settings
from copperchat.errors import StructuredToolError
from copperchat.kicad_session import DocKind, KicadSession
from copperchat.tools.spec import ToolContext, ToolResult

_UNITS_TO_MM = 1.0   # kicad-cli JSON is mm (spike 1C verifies; if it records otherwise, scale here)


def _pos(p: dict[str, Any] | None) -> dict[str, float] | None:
    if not p:
        return None
    return {"x": round(float(p["x"]) * _UNITS_TO_MM, 4), "y": round(float(p["y"]) * _UNITS_TO_MM, 4)}


def _entry(e: dict[str, Any], default_type: str) -> dict[str, Any]:
    items = [{"uuid": str(i.get("uuid", "")), "description": str(i.get("description", "")),
              "position_mm": _pos(i.get("pos"))} for i in e.get("items", [])]
    return {"type": str(e.get("type", default_type)), "severity": str(e.get("severity", "error")),
            "description": str(e.get("description", "")),
            "position_mm": items[0]["position_mm"] if items else None, "items": items}


def parse_violations(raw: dict[str, Any], doc: DocKind) -> list[dict[str, Any]]:
    if doc == "sch":
        return [_entry(v, "erc") for s in raw.get("sheets", []) for v in s.get("violations", [])]
    out = [_entry(v, "drc") for v in raw.get("violations", [])]
    out += [_entry(v, "unconnected_items") for v in raw.get("unconnected_items", [])]
    out += [_entry(v, "schematic_parity") for v in raw.get("schematic_parity", [])]
    return out


async def run_kicad_check(session: KicadSession, doc: DocKind, settings: Settings) -> str:
    """Saves a copy via the API, runs kicad-cli ERC/DRC on it, returns the JSON text."""
    with tempfile.TemporaryDirectory(prefix="cc-check-") as tmp:
        stem = session.project.name
        src = await session.save_copy(doc, Path(tmp) / f"{stem}.kicad_{'sch' if doc == 'sch' else 'pcb'}")
        if doc == "sch":
            await session.save_copy("pcb", Path(tmp) / f"{stem}.kicad_pcb")
        report = Path(tmp) / "report.json"
        sub = ["sch", "erc"] if doc == "sch" else ["pcb", "drc", "--schematic-parity"]
        proc = await asyncio.create_subprocess_exec(
            settings.kicad_cli, *sub, "--format", "json", "--severity-all",
            "--output", str(report), str(src),
            env={**os.environ, "LANG": "C", "LC_ALL": "C"},
            stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE)
        try:
            _, err = await asyncio.wait_for(proc.communicate(), settings.kicad_api_timeout_s * 4)
        except TimeoutError:
            proc.kill()
            raise StructuredToolError(f"{'ERC' if doc == 'sch' else 'DRC'} timed out",
                                      "kicad-cli did not finish in time",
                                      "Retry; if it persists, render the view instead") from None
        if not report.exists():
            raise StructuredToolError(f"{'ERC' if doc == 'sch' else 'DRC'} failed",
                                      err.decode(errors="replace")[-500:] or "no report written",
                                      "Check that the document is open and saved")
        return report.read_text()


async def _check(ctx: ToolContext, doc: DocKind) -> ToolResult:
    raw = json.loads(await run_kicad_check(ctx.session, doc, ctx.settings))
    vs = parse_violations(raw, doc)
    return ToolResult(data={"count": len(vs), "violations": vs})


async def run_erc(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    return await _check(ctx, "sch")


async def run_drc(ctx: ToolContext, args: dict[str, Any]) -> ToolResult:
    return await _check(ctx, "pcb")

from __future__ import annotations

import shutil
from pathlib import Path

from copperchat.kicad_session import DocKind, KicadSession
from copperchat.projects import Project, agent_dir

_EXT = {"sch": "kicad_sch", "pcb": "kicad_pcb"}


def snapshot_path(project: Project, doc: DocKind) -> Path:
    return agent_dir(project) / "view" / f"{project.name}.{_EXT[doc]}"


async def write_kicanvas_snapshot(session: KicadSession, doc: DocKind) -> Path:
    dest = snapshot_path(session.project, doc)
    await session.save_copy(doc, dest)
    if doc == "sch":
        # KiCanvas needs sub-sheets next to the root. If spike 1C records that
        # SaveCopyOfDocument already writes them, this loop finds nothing to do.
        sheets = await session.call(lambda h: [str(s.filename) for s in h.schematic().get_hierarchy()
                                               if getattr(s, "filename", "")])
        for name in sheets:
            src = session.project.dir / name
            if src.is_file() and not (dest.parent / name).exists():
                shutil.copy2(src, dest.parent / name)   # byte copy of KiCad's own file
    session.view_version += 1
    return dest

from __future__ import annotations

import re
from typing import Any

from fastapi import APIRouter, Depends, Query, Request, Response
from fastapi.responses import FileResponse

from copperchat.errors import CopperChatError
from copperchat.projects import agent_dir, get_project
from copperchat.views.snapshot import snapshot_path
from copperchat.web.auth import User, require_user

router = APIRouter(prefix="/api/projects", tags=["views"])
_RENDER = re.compile(r"^[0-9a-f]{32}\.png$")
_SHEET = re.compile(r"^[A-Za-z0-9_. -]+\.kicad_sch$")


class BadViewRequest(CopperChatError):
    code = "bad_view_request"
    http_status = 400


class NotFoundError(CopperChatError):
    code = "not_found"
    http_status = 404


def _session_for(request: Request, project_id: str) -> Any:
    project = get_project(request.app.state.settings.projects_root, project_id)
    current = request.app.state.sessions.current()
    if current is None or current.project.id != project.id:
        raise BadViewRequest("project is not open - POST /api/projects/{id}/open first")
    return current


@router.get("/{project_id}/view/{doc}")
async def get_view(project_id: str, doc: str, request: Request,
                   _: User = Depends(require_user)) -> Response:
    if doc not in ("sch", "pcb"):
        raise BadViewRequest("doc must be 'sch' or 'pcb'")
    session = _session_for(request, project_id)
    path = snapshot_path(session.project, doc)  # type: ignore[arg-type]
    if not path.is_file():
        raise NotFoundError("no view snapshot yet")
    etag = f'"{session.view_version}"'
    if request.headers.get("if-none-match") == etag:
        return Response(status_code=304, headers={"ETag": etag})
    return FileResponse(path, media_type="application/octet-stream",
                        headers={"ETag": etag, "Cache-Control": "no-cache"})


@router.get("/{project_id}/view/sheets/{file}")
async def get_sheet(project_id: str, file: str, request: Request,
                    _: User = Depends(require_user)) -> Response:
    if not _SHEET.match(file):
        raise BadViewRequest("invalid sheet file name")
    session = _session_for(request, project_id)
    path = agent_dir(session.project) / "view" / file
    if not path.is_file():
        raise NotFoundError("no such sheet")
    return FileResponse(path, media_type="application/octet-stream",
                        headers={"ETag": f'"{session.view_version}"', "Cache-Control": "no-cache"})


@router.get("/{project_id}/renders/{file}")
async def get_render(project_id: str, file: str, request: Request,
                     _: User = Depends(require_user)) -> Response:
    if not _RENDER.match(file):
        raise BadViewRequest("invalid render file name")
    project = get_project(request.app.state.settings.projects_root, project_id)
    root = (agent_dir(project) / "renders").resolve()
    path = (root / file).resolve()
    if path.parent != root or not path.is_file():
        raise NotFoundError("no such render")
    return FileResponse(path, media_type="image/png")


@router.get("/{project_id}/agent-view")
async def get_agent_view(project_id: str, request: Request, doc: str = Query("pcb"),
                         theme: str = Query("agent"), layers: str | None = Query(None),
                         _: User = Depends(require_user)) -> Response:
    from copperchat.views.render import render_view

    if doc not in ("sch", "pcb") or theme not in ("agent", "light", "dark"):
        raise BadViewRequest("doc must be sch|pcb, theme agent|light|dark")
    session = _session_for(request, project_id)
    layer_list = [x for x in (layers or "").split(",") if x] or None
    img = await render_view(session, doc, "agent", layers=layer_list, theme=theme)  # type: ignore[arg-type]
    return Response(content=img.data, media_type="image/png",
                    headers={"Cache-Control": "no-cache"})

from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from copperchat.errors import StructuredToolError
from copperchat.kicad_tools.filtering import not_found


@dataclass(frozen=True)
class Region:
    x_mm: float | None = None
    y_mm: float | None = None
    w_mm: float | None = None
    h_mm: float | None = None
    around_ref: str | None = None
    margin_mm: float = 5.0


def resolve_region(region: Region | None,
                   snap: dict[str, Any]) -> tuple[float, float, float, float] | None:
    if region is None:
        return None
    if region.around_ref:
        fps = snap["footprints"]
        hit = next((f for f in fps if f["ref"] == region.around_ref), None)
        if hit is None:
            raise not_found("Footprint", region.around_ref, (f["ref"] for f in fps))
        b, m = hit["bbox_mm"], region.margin_mm
        return (b["x"] - m, b["y"] - m, b["w"] + 2 * m, b["h"] + 2 * m)
    box = (region.x_mm, region.y_mm, region.w_mm, region.h_mm)
    if any(v is None for v in box) or (region.w_mm or 0) <= 0 or (region.h_mm or 0) <= 0:
        raise StructuredToolError("Invalid region", "a mm box needs x, y, w>0 and h>0",
                                  "Pass x_mm, y_mm, w_mm, h_mm or around_ref")
    return (float(region.x_mm), float(region.y_mm), float(region.w_mm), float(region.h_mm))  # type: ignore[arg-type]

def _bounds(snap: dict[str, Any]) -> tuple[float, float, float, float]:
    xs: list[float] = []
    ys: list[float] = []
    for s in snap["outline"]:
        xs += [s["start_mm"]["x"], s["end_mm"]["x"]]
        ys += [s["start_mm"]["y"], s["end_mm"]["y"]]
    for f in snap["footprints"]:
        b = f["bbox_mm"]
        xs += [b["x"], b["x"] + b["w"]]
        ys += [b["y"], b["y"] + b["h"]]
    if not xs:
        return (0.0, 0.0, 10.0, 10.0)
    return (min(xs) - 3, min(ys) - 3, max(xs) - min(xs) + 6, max(ys) - min(ys) + 6)


def _grid_step(w_mm: float) -> float:
    for step in (1, 2, 5, 10, 20, 50):
        if w_mm / step <= 20:
            return float(step)
    return 100.0

    x0, y0, w, h = region_box or _bounds(snap)
    scale = min(px_per_mm, MAX_PX / max(w, h))
    img = Image.new("RGB", (max(1, round(w * scale)), max(1, round(h * scale))), t["bg"])
    d = ImageDraw.Draw(img)
    font = ImageFont.load_default(size=max(10, round(scale * 0.9)))

    def p(pt: dict[str, float]) -> tuple[float, float]:
        return ((pt["x"] - x0) * scale, (pt["y"] - y0) * scale)

    step = _grid_step(w)
    gx = math.ceil(x0 / step) * step
    while gx < x0 + w:
        d.line([p({"x": gx, "y": y0}), p({"x": gx, "y": y0 + h})], fill=t["grid"])
        d.text(p({"x": gx, "y": y0}), f"{gx:g}", fill=t["text"], font=font)
        gx += step
    gy = math.ceil(y0 / step) * step
    while gy < y0 + h:
        d.line([p({"x": x0, "y": gy}), p({"x": x0 + w, "y": gy})], fill=t["grid"])
        d.text(p({"x": x0, "y": gy}), f"{gy:g}", fill=t["text"], font=font)
        gy += step

    for s in snap["outline"]:
        d.line([p(s["start_mm"]), p(s["end_mm"])], fill=t["outline"], width=2)
    for tr in snap["tracks"]:
        if layers and tr["layer"] not in layers:
            continue
        d.line([p(tr["start_mm"]), p(tr["end_mm"])], fill=net_color(tr["net"], theme),
               width=max(1, round(tr["width_mm"] * scale)))
    for v in snap["vias"]:
        cx, cy = p(v["position_mm"])
        r = 0.4 * scale
        d.ellipse([cx - r, cy - r, cx + r, cy + r], fill=net_color(v["net"], theme))

    taken: list[tuple[float, float, float, float]] = []
    for f in snap["footprints"]:
        if layers and not any(lay.startswith("F.") if f["side"] == "front" else lay.startswith("B.")
                              for lay in layers):
            continue
        b = f["bbox_mm"]
        d.rectangle([p({"x": b["x"], "y": b["y"]}), p({"x": b["x"] + b["w"], "y": b["y"] + b["h"]})],
                    outline=t["fp"])
        for pad in f["pads"]:
            cx, cy = p(pad["position_mm"])
            r = 0.35 * scale
            col = net_color(pad["net"], theme) if pad["net"] else t["fp"]
            d.rectangle([cx - r, cy - r, cx + r, cy + r], fill=col)
        label = f"{f['ref']} ({f['position_mm']['x']:g},{f['position_mm']['y']:g})"
        lx, ly = p({"x": b["x"], "y": b["y"]})
        tw, th = d.textbbox((0, 0), label, font=font)[2:]
        box = (lx, ly - th - 2, lx + tw, ly - 2)
        while any(not (box[2] < o[0] or box[0] > o[2] or box[3] < o[1] or box[1] > o[3])
                  for o in taken) and box[1] > -h * scale:
            box = (box[0], box[1] - th - 1, box[2], box[3] - th - 1)   # de-clutter: shift up
        taken.append(box)
        d.text((box[0], box[1]), label, fill=t["text"], font=font)

    for u in snap["unconnected"]:
        d.line([p(u["from_mm"]), p(u["to_mm"])], fill=t["air"], width=1)
    for m in markers:
        if not m.get("position_mm"):
            continue
        cx, cy = p(m["position_mm"])
        col = t["err"] if m.get("severity") == "error" else t["warn"]
        r = 0.8 * scale
        d.polygon([(cx, cy - r), (cx + r, cy + r), (cx - r, cy + r)], outline=col, width=2)
    buf = io.BytesIO()
    img.save(buf, format="PNG", optimize=False)
    return buf.getvalue()


def overlay_grid(png: bytes, origin_mm: tuple[float, float], px_per_mm: float,
                 markers: list[dict[str, Any]], theme: str) -> bytes:
    """mm grid + ERC markers over a KiCad-exported schematic PNG with a known mapping."""
    t = THEMES.get(theme, THEMES["agent"])
    img = Image.open(io.BytesIO(png)).convert("RGB")
    d = ImageDraw.Draw(img, "RGBA")
    w_mm, h_mm = img.width / px_per_mm, img.height / px_per_mm
    step = _grid_step(w_mm)
    font = ImageFont.load_default(size=12)
    x = math.ceil(origin_mm[0] / step) * step
    while x < origin_mm[0] + w_mm:
        px = (x - origin_mm[0]) * px_per_mm
        d.line([(px, 0), (px, img.height)], fill=(*t["grid"], 160))
        d.text((px + 2, 2), f"{x:g}", fill=t["err"], font=font)
        x += step
    y = math.ceil(origin_mm[1] / step) * step
    while y < origin_mm[1] + h_mm:
        py = (y - origin_mm[1]) * px_per_mm
        d.line([(0, py), (img.width, py)], fill=(*t["grid"], 160))
        d.text((2, py + 2), f"{y:g}", fill=t["err"], font=font)
        y += step
    for m in markers:
        if m.get("position_mm"):
            cx = (m["position_mm"]["x"] - origin_mm[0]) * px_per_mm
            cy = (m["position_mm"]["y"] - origin_mm[1]) * px_per_mm
            d.ellipse([cx - 10, cy - 10, cx + 10, cy + 10], outline=t["err"], width=3)
    buf = io.BytesIO()
    img.save(buf, format="PNG")
    return buf.getvalue()

def assert_image_close(actual: bytes, golden: Path, *, threshold: int = 16,
                       max_ratio: float = 0.01) -> None:
    """Fails if more than max_ratio of pixels differ by more than threshold (any channel)."""
    if os.environ.get("UPDATE_GOLDENS") == "1":
        golden.parent.mkdir(parents=True, exist_ok=True)
        golden.write_bytes(actual)
        return
    a = Image.open(io.BytesIO(actual)).convert("RGB")
    g = Image.open(golden).convert("RGB")
    assert a.size == g.size, f"size {a.size} != golden {g.size} ({golden.name})"
    diff = ImageChops.difference(a, g).convert("L").point(lambda v: 255 if v > threshold else 0)
    bad = sum(1 for v in diff.getdata() if v)
    ratio = bad / (a.width * a.height)
    if ratio > max_ratio:
        out = golden.with_suffix(".actual.png")
        out.write_bytes(actual)
        raise AssertionError(f"{golden.name}: {ratio:.3%} pixels differ (> {max_ratio:.1%}); "
                             f"wrote {out.name}; run scripts/update-goldens.sh if intended")

# Set True only if docs/spikes/1c-api-server.md records the PNG pixel->mm mapping.
PIXEL_MAPPING_KNOWN = False
# Set True if 1C records RunSchematicJobExportPng / RunBoardJobExportPng at the pinned commit.
API_PNG_EXPORT = False


async def _cli_svg_to_png(session: KicadSession, doc: DocKind, theme: str, tmp: Path) -> bytes:
    import cairosvg

    ext = "kicad_sch" if doc == "sch" else "kicad_pcb"
    src = await session.save_copy(doc, tmp / f"{session.project.name}.{ext}")
    out = tmp / "out"
    out.mkdir()
    args = (["sch", "export", "svg", "--exclude-drawing-sheet", "--output", str(out)]
            if doc == "sch" else
            ["pcb", "export", "svg", "--mode-single", "--exclude-drawing-sheet",
             "--layers", "F.Cu,B.Cu,F.SilkS,Edge.Cuts", "--output", str(out / "board.svg")])
    if theme == "dark":
        args += ["--theme", "_builtin_default"]   # 1C records the dark theme name if different
    proc = await asyncio.create_subprocess_exec(
        session.settings.kicad_cli, *args, str(src),
        env={**os.environ, "LANG": "C", "LC_ALL": "C"},
        stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE)
    _, err = await asyncio.wait_for(proc.communicate(), session.settings.kicad_api_timeout_s * 4)
    svgs = sorted(out.glob("*.svg"))
    if not svgs:
        raise StructuredToolError("KiCad export failed", err.decode(errors="replace")[-500:],
                                  "Use render_view with mode=agent")
    bg = "white" if theme != "dark" else "black"
    return bytes(cairosvg.svg2png(url=str(svgs[0]), output_width=2000, background_color=bg))


async def export_kicad_png(session: KicadSession, doc: DocKind,
                           theme: str) -> tuple[bytes, tuple[float, float] | None, float | None]:
    with tempfile.TemporaryDirectory(prefix="cc-render-") as t:
        tmp = Path(t)
        if API_PNG_EXPORT:
            # If docs/spikes/1c-api-server.md records a different call, use that.
            dest = tmp / "out.png"
            await session.call(lambda h: (h.schematic() if doc == "sch" else h.board())
                               .export_png(str(dest)))  # type: ignore[attr-defined]
            return dest.read_bytes(), None, None
        return await _cli_svg_to_png(session, doc, theme, tmp), None, None

from __future__ import annotations

import json
import uuid
from typing import Literal

from copperchat.errors import StructuredToolError
from copperchat.kicad_session import DocKind, KicadSession
from copperchat.kicad_tools.checks import parse_violations, run_kicad_check
from copperchat.kicad_tools.pcb import board_snapshot
from copperchat.projects import agent_dir
from copperchat.tools.spec import ImageBlock
from copperchat.views.agent_render import overlay_grid, render_board
from copperchat.views.kicad_render import PIXEL_MAPPING_KNOWN, export_kicad_png
from copperchat.views.region import Region, resolve_region


async def render_view(session: KicadSession, doc: DocKind, mode: Literal["agent", "kicad"],
                      region: Region | None = None, layers: list[str] | None = None,
                      theme: Literal["agent", "light", "dark"] = "agent") -> ImageBlock:
    if region is not None and (mode == "kicad" or doc == "sch") and not PIXEL_MAPPING_KNOWN:
        raise StructuredToolError("Region crop not available here",
                                  "region crop is supported in agent mode for the board only",
                                  "Use mode=agent with doc=pcb, or omit region")
    if mode == "agent" and doc == "pcb":
        snap = await session.call(board_snapshot)
        box = resolve_region(region, snap)
        markers = parse_violations(json.loads(await run_kicad_check(session, "pcb",
                                                                    session.settings)), "pcb")
        data = render_board(snap, region_box=box, layers=layers, theme=theme, markers=markers)
    else:
        png, origin, ppm = await export_kicad_png(session, doc, "dark" if theme == "dark" else "light")
        data = png
        if mode == "agent" and origin is not None and ppm is not None:
            markers = parse_violations(json.loads(await run_kicad_check(session, "sch",
                                                                        session.settings)), "sch")
            data = overlay_grid(png, origin, ppm, markers, theme)
    out_dir = agent_dir(session.project) / "renders"
    out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / f"{uuid.uuid4().hex}.png"
    path.write_bytes(data)
    return ImageBlock(mime="image/png", data=data, path=path)

from copperchat.views.region import Region
